program Test2;

struct Vector2 {
    let x: float;
    let y: float;
}

function scale(v:Vector2, s:float): Vector2 {
    let r:Vector2;
    r.x = v.x * s;
    r.y = v.y * s;
    return r;
}

main {
    let v:Vector2;
    v.x = 1.5;
    v.y = -3.0;

    let w:Vector2;
    w = scale(v, 2.0);

    print(w.x);
    print(w.y);
}
